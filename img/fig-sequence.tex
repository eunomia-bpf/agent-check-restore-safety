\documentclass[border=10pt]{standalone}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,positioning,calc}
\begin{document}
\begin{tikzpicture}[font=\Large, >=Stealth]

% Entities
\node[font=\huge\bfseries] (user) at (0,0) {User};
\node[font=\huge\bfseries] (agent) at (5,0) {Agent};
\node[font=\huge\bfseries] (bank) at (10,0) {Bank};
\node[font=\huge\bfseries] (bob) at (15,0) {Bob's MCP};

% Lifelines
\draw[dashed, gray] (user) -- (0,-10.5);
\draw[dashed, gray] (agent) -- (5,-10.5);
\draw[dashed, gray] (bank) -- (10,-10.5);
\draw[dashed, gray] (bob) -- (15,-10.5);

% Step 1: User sends task
\draw[->, thick] (0,-1.2) -- node[above] {``Transfer \$500 to Bob''} (5,-1.2);

% Checkpoint marker
\draw[dashed, blue!60!black, very thick] (-0.8,-1.6) -- (15.8,-1.6);
\node[fill=blue!10, draw=blue!50!black, rounded corners, font=\large\bfseries] at (7.5,-1.6) {CHECKPOINT};

% Step 2: Agent calls bank
\draw[->, thick, blue!70!black] (5,-2.8) -- node[above] {transfer(ref=\texttt{a1b2c3d4})} (10,-2.8);

% Step 3: Bank responds
\draw[<-, thick, green!50!black] (5,-3.7) -- node[above] {Success, balance=\$9500} (10,-3.7);

% Step 4: Agent calls Bob's service
\draw[->, thick] (5,-4.7) -- node[above] {confirm\_receipt()} (15,-4.7);

% Step 5: Bob crashes the agent
\draw[<-, thick, red!80!black] (5,-5.7) -- node[above, red!80!black] {malformed response $\rightarrow$ CRASH} (15,-5.7);

% Restore marker
\draw[dashed, red!60!black, very thick] (-0.8,-6.8) -- (15.8,-6.8);
\node[fill=yellow!30, draw=red!50!black, rounded corners, font=\large\bfseries, text=red!70!black] at (7.5,-6.8) {RESTORE to CHECKPOINT};

% Restore arrow
\draw[->, very thick, red!50!black, dashed] (3.8,-6.8) .. controls (2.3,-4.5) and (2.3,-3) .. (3.8,-1.6);

% Step 6: Agent re-attempts with different ref
\draw[->, thick, red!70!black] (5,-8.1) -- node[above] {transfer(ref=\texttt{f9a8b7c6})} (10,-8.1);

% Callout: ref changed
\node[draw=red!70!black, fill=red!10, rounded corners, align=center] (callout) at (13.2,-8.1) {ref changed!\\[-2pt]\texttt{a1b2..}$\rightarrow$\texttt{f9a8..}};
\draw[->, red, thick] (callout.west) -- (10.2,-8.1);

% Step 7: Bank accepts as new
\draw[<-, thick, red!70!black] (5,-9.1) -- node[above] {Success, balance=\$9000} (10,-9.1);

% Result + state annotations in one row
\node[fill=red!20, draw=red!60!black, rounded corners, align=center] at (2,-11) {\textbf{Result:}\\[1pt]\$1000 transferred\\(Bob gets 2$\times$)};
\node[draw, fill=blue!5, rounded corners, align=center] at (7.5,-11) {\textbf{Agent's view:}\\[1pt]``I transferred once''};
\node[draw, fill=red!5, rounded corners, align=center] at (13.2,-11) {\textbf{Bank's reality:}\\[1pt]Two transactions\\with different IDs};

\end{tikzpicture}
